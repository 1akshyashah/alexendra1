┃ 9. TROUBLESHOOTING                                                        ┃
┣━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┫
┃                                                                            ┃
┃  Loss is NaN?                                                              ┃
┃    → Reduce LEARNING_RATE 10x (0.1 → 0.01)                               ┃
┃                                                                            ┃
┃  Out of Memory?                                                            ┃
┃    → Reduce BATCH_SIZE to 128 or 64                                       ┃
┃    → Enable USE_AMP = True                                                ┃
┃                                                                            ┃
┃  Training too slow?                                                        ┃
┃    → Increase NUM_WORKERS to 8-16                                         ┃
┃    → Enable USE_AMP = True                                                ┃
┃                                                                            ┃
┃  Overfitting?                                                              ┃
┃    → Increase WEIGHT_DECAY from 5e-4 to 1e-3                             ┃
┃    → Increase LABEL_SMOOTHING from 0.1 to 0.2                            ┃
┃                                                                            ┃
